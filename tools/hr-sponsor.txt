create table if not exists public.hr_emp_info ( emp_id uuid primary key references public.employees(id) on delete cascade, sponsor text check (sponsor in ('ours','other','none')), sponsor_name text not null default '', updated_at timestamptz not null default now(), updated_by text not null default '' ); create or replace function public.hr_info_stamp() returns trigger language plpgsql security definer set search_path = public as $$ begin new.sponsor_name := left(btrim(coalesce(new.sponsor_name, '')), 80); if new.sponsor is distinct from 'other' then new.sponsor_name := ''; end if; new.updated_at := now(); new.updated_by := hr_who(); return new; end $$; drop trigger if exists hr_info_stamp on public.hr_emp_info; create trigger hr_info_stamp before insert or update on public.hr_emp_info for each row execute function public.hr_info_stamp(); alter table public.hr_emp_info enable row level security; drop policy if exists hri_sel on public.hr_emp_info; drop policy if exists hri_ins on public.hr_emp_info; drop policy if exists hri_upd on public.hr_emp_info; drop policy if exists hri_del on public.hr_emp_info; create policy hri_sel on public.hr_emp_info for select to authenticated using ( hr_can() or exists (select 1 from employees e where e.id = emp_id and employees_can(e.branch))); create policy hri_ins on public.hr_emp_info for insert to authenticated with check (hr_can()); create policy hri_upd on public.hr_emp_info for update to authenticated using (hr_can()) with check (hr_can()); create policy hri_del on public.hr_emp_info for delete to authenticated using (is_owner()); revoke all on public.hr_emp_info from anon; grant select, insert, update, delete on public.hr_emp_info to authenticated; select 'تم تفعيل حالة الكفالة ✓' as result;
