select p.username, p.full_name, p.role, p.created_at::date as created, exists (select 1 from auth.users u where u.id = p.id) as login_exists, (select e.name from employees e where regexp_replace(coalesce(e.phone,''),'\D','','g') like '%' || right(regexp_replace(p.username,'\D','','g'), 9) and length(regexp_replace(p.username,'\D','','g')) >= 9 limit 1) as matching_employee from profiles p where p.role = 'employee' and not exists (select 1 from emp_accounts a where a.user_id = p.id) order by p.created_at desc;
