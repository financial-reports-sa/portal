create or replace function public.pr_edit(p_id uuid, p_title text, p_items jsonb default '[]', p_details text default null, p_priority text default 'normal', p_need date default null) returns void language plpgsql security definer set search_path = public as $$ declare m purch_requests; w text := pr_who(); it jsonb; begin select * into m from purch_requests where id = p_id for update; if not found then raise exception 'الطلب غير موجود'; end if; if m.requested_uid is distinct from auth.uid() then raise exception 'التعديل لصاحب الطلب فقط'; end if; if m.status <> 'new' then raise exception 'ما يمكن التعديل — المشتريات استلمت الطلب'; end if; if m.created_at < now() - interval '12 hours' then raise exception 'انتهت مهلة التعديل (12 ساعة من رفع الطلب)'; end if; if coalesce(trim(p_title), '') = '' then raise exception 'اكتب عنوان الطلب'; end if; select coalesce(jsonb_agg(jsonb_build_object('n', left(btrim(x->>'n'), 80), 'q', round(greatest((x->>'q')::numeric, 0), 2), 'u', left(coalesce(x->>'u', ''), 20))), '[]') into it from (select x from jsonb_array_elements(case when jsonb_typeof(p_items) = 'array' then p_items else '[]' end) x limit 40) s where coalesce(btrim(x->>'n'), '') <> '' and coalesce((x->>'q')::numeric, 0) > 0; if m.kind = 'items' and jsonb_array_length(it) = 0 then raise exception 'أضف صنف واحد على الأقل'; end if; update purch_requests set title = left(trim(p_title), 120), items = case when m.kind = 'items' then it else items end, details = nullif(left(trim(coalesce(p_details, '')), 800), ''), priority = case when p_priority = 'urgent' then 'urgent' else 'normal' end, need_by = p_need, updated_at = now(), log = log || jsonb_build_array(jsonb_build_object('t', now(), 'by', w, 'a', 'edit')) where id = p_id; begin perform push_send(pr_sups(), '✏️ تعديل طلب شراء · ' || coalesce((select name from branches where id = m.branch), m.branch), 'PR-' || lpad(m.no::text, 5, '0') || ' · ' || left(trim(p_title), 80) || ' — ' || coalesce(w, ''), './?tab=purch', 'pr-' || p_id); exception when others then null; end; end $$; create or replace function public.pr_list() returns jsonb language sql stable security definer set search_path = public as $$ select coalesce(jsonb_agg(jsonb_build_object('id', x.id, 'no', x.no, 'branch', x.branch, 'kind', x.kind, 'title', x.title, 'items', x.items, 'details', x.details, 'priority', x.priority, 'need_by', x.need_by, 'status', x.status, 'requested_by', x.requested_by, 'created_at', x.created_at, 'started_at', x.started_at, 'bought_at', x.bought_at, 'closed_at', x.closed_at, 'closed_by', x.closed_by, 'handled_by', x.handled_by, 'supplier', x.supplier, 'cost', x.cost, 'sup_note', x.sup_note, 'reject_reason', x.reject_reason, 'has_photo', (x.photo is not null or x.inv_photo is not null), 'log', x.log, 'can_edit', (x.requested_uid = auth.uid() and x.status = 'new' and x.created_at > now() - interval '12 hours'), 'edit_until', x.created_at + interval '12 hours') order by x.created_at desc), '[]') from (select * from purch_requests p where pr_role() in ('admin','sup') or (pr_role() = 'mgr' and p.branch = any(my_branches())) order by created_at desc limit 500) x $$; revoke all on function public.pr_edit(uuid, text, jsonb, text, text, date) from public, anon; grant execute on function public.pr_edit(uuid, text, jsonb, text, text, date) to authenticated; revoke all on function public.pr_list() from public, anon; grant execute on function public.pr_list() to authenticated; select 'تم: تعديل طلب الشراء خلال 12 ساعة' as result;
