select 'الموظف: ' || e.name || ' | الحالة: ' || e.status || ' | رقم الدخول: ' || coalesce(a.phone, 'لا يوجد حساب') || ' | الحساب موجود: ' || case when a.user_id is null then '-' when exists (select 1 from auth.users u where u.id = a.user_id) then 'نعم' else 'لا' end || ' | النوع: ' || coalesce(p.role, '-') || ' | مفعل: ' || coalesce(p.active::text, '-') || ' | اسم المستخدم: ' || coalesce(p.username, '-') as info from employees e left join emp_accounts a on a.emp_id = e.id left join profiles p on p.id = a.user_id where regexp_replace(coalesce(e.phone, ''), '\D', '', 'g') like '%568215144' or e.name like '%اسلام%فتحي%' or e.name like '%إسلام%فتحي%' or a.phone like '%568215144%';
