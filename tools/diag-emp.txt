select e.id as emp_id, e.name, e.branch, e.status, e.phone as emp_phone, a.phone as login_phone, a.user_id, p.username, p.role, p.active, exists (select 1 from auth.users u where u.id = a.user_id) as login_exists, (select count(*) from emp_accounts z where z.user_id = a.user_id) as links_for_user from employees e left join emp_accounts a on a.emp_id = e.id left join profiles p on p.id = a.user_id where regexp_replace(coalesce(e.phone, ''), '\D', '', 'g') like '%568215144' or e.name like '%اسلام%فتحي%' or e.name like '%إسلام%فتحي%' or a.phone like '%568215144%';
