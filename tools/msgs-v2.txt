create or replace function public.msg_inbox() returns jsonb language sql stable security definer set search_path = public as $$ select coalesce(jsonb_agg(x order by x->>'at' desc), '[]') from ( select jsonb_build_object('id', m.id, 'kind', m.kind, 'prio', m.prio, 'title', m.title, 'body', left(m.body, 160), 'ack', m.need_ack, 'from', m.sender_name, 'at', m.created_at, 'read', r.read_at, 'acked', r.ack_at, 'files', jsonb_array_length(m.files), 'auto', m.target->>'auto') x from msg_rcpt r join msgs m on m.id = r.msg_id where r.user_id = auth.uid() order by m.id desc limit 300) t $$; create or replace function public.msg_sent() returns jsonb language sql stable security definer set search_path = public as $$ select coalesce(jsonb_agg(x order by x->>'at' desc), '[]') from ( select jsonb_build_object('id', m.id, 'kind', m.kind, 'prio', m.prio, 'title', m.title, 'ack', m.need_ack, 'from', m.sender_name, 'mine', m.sender = auth.uid(), 'at', m.created_at, 'target', m.target, 'auto', m.target->>'auto', 'to', (select string_agg(coalesce(r.name, ''), '، ' order by case when r.kind = 'emp' then 0 when r.kind = 'mgr' then 1 else 2 end) from msg_rcpt r where r.msg_id = m.id), 'n', (select count(*) from msg_rcpt r where r.msg_id = m.id), 'nr', (select count(*) from msg_rcpt r where r.msg_id = m.id and r.read_at is not null), 'na', (select count(*) from msg_rcpt r where r.msg_id = m.id and r.ack_at is not null), 'nrep', (select count(*) from msg_replies q where q.msg_id = m.id and q.user_id <> m.sender)) x from msgs m where m.sender = auth.uid() or coalesce(is_admin(), false) order by m.id desc limit 400) t $$; create or replace function public.msg_read_all() returns int language plpgsql volatile security definer set search_path = public as $$ declare n int; begin update msg_rcpt r set read_at = now() from msgs m where m.id = r.msg_id and r.user_id = auth.uid() and r.read_at is null and not m.need_ack; get diagnostics n = row_count; return n; end $$; grant execute on function public.msg_read_all() to authenticated;
