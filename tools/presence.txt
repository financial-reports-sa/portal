create table if not exists public.user_presence (   user_id uuid primary key references auth.users(id) on delete cascade,   last_seen timestamptz not null default now(),   login_at timestamptz,   out_at timestamptz,   out_reason text not null default '',   tab text not null default '',   device text not null default '' ); alter table public.user_presence enable row level security; revoke all on public.user_presence from anon, authenticated;  create or replace function public.presence_ping(p_tab text default '', p_device text default '', p_login boolean default false) returns void language plpgsql security definer set search_path = public as $$ begin   if auth.uid() is null then return; end if;   if exists (select 1 from profiles where id = auth.uid() and role = 'employee') then return; end if;   insert into user_presence (user_id, last_seen, login_at, out_at, out_reason, tab, device)   values (auth.uid(), now(), case when p_login then now() end, null, '', left(coalesce(p_tab, ''), 40), left(coalesce(p_device, ''), 80))   on conflict (user_id) do update set last_seen = now(), out_at = null, out_reason = '',     login_at = case when p_login or user_presence.out_at is not null then now() else coalesce(user_presence.login_at, now()) end,     tab = left(coalesce(nullif(p_tab, ''), user_presence.tab), 40), device = left(coalesce(nullif(p_device, ''), user_presence.device), 80); end $$;  create or replace function public.presence_out(p_reason text default 'logout') returns void language plpgsql security definer set search_path = public as $$ begin   if auth.uid() is null then return; end if;   update user_presence set out_at = now(), out_reason = case when p_reason in ('logout', 'idle') then p_reason else 'logout' end where user_id = auth.uid(); end $$;  create or replace function public.presence_list() returns jsonb language sql stable security definer set search_path = public as $$   select case when not coalesce(is_admin(), false) then '[]'::jsonb else coalesce((     select jsonb_agg(jsonb_build_object('id', p.id, 'name', coalesce(nullif(p.full_name, ''), p.username), 'username', p.username, 'role', p.role, 'active', p.active,       'last_seen', x.last_seen, 'login_at', x.login_at, 'out_at', x.out_at, 'out_reason', x.out_reason, 'tab', x.tab, 'device', x.device) order by x.last_seen desc nulls last)     from profiles p left join user_presence x on x.user_id = p.id     where p.role in ('user', 'admin', 'owner') and coalesce(p.active, true)), '[]'::jsonb) end $$;
