create table if not exists public.growth_shifts ( branch     text not null, shift_id   text not null, bdate      date not null, user_id    text not null default '', cashier    text not null default '', opened     timestamptz, closed     timestamptz, bills      int not null default 0, net        numeric(14,2) not null default 0, cash       numeric(14,2) not null default 0, card       numeric(14,2) not null default 0, credit     numeric(14,2) not null default 0, refunds    numeric(14,2) not null default 0, apps       jsonb not null default '{}', synced_at  timestamptz not null default now(), primary key (branch, shift_id) ); create index if not exists growth_shifts_day_idx on public.growth_shifts (branch, bdate); alter table public.growth_shifts enable row level security; revoke all on public.growth_shifts from anon, authenticated; create table if not exists public.growth_cashier_map ( branch   text not null, user_id  text not null, code     text not null, at       timestamptz not null default now(), primary key (branch, user_id) ); alter table public.growth_cashier_map enable row level security; revoke all on public.growth_cashier_map from anon, authenticated; create or replace function public.growth_ingest_shifts(p_rows jsonb, p_days date[] default null) returns jsonb language plpgsql volatile security definer set search_path = public as $$ declare n int := 0; end if; if p_days is not null then delete from growth_shifts g where g.bdate = any(p_days) and not exists (select 1 from jsonb_array_elements(p_rows) x where x->>'branch' = g.branch and x->>'shift_id' = g.shift_id); end if; insert into growth_shifts (branch, shift_id, bdate, user_id, cashier, opened, closed, bills, net, cash, card, credit, refunds, apps, synced_at) select x->>'branch', x->>'shift_id', (x->>'bdate')::date, left(coalesce(x->>'user_id', ''), 40), left(coalesce(x->>'cashier', ''), 80), nullif(x->>'opened', '')::timestamptz, nullif(x->>'closed', '')::timestamptz, coalesce((x->>'bills')::int, 0), coalesce((x->>'net')::numeric, 0), coalesce((x->>'cash')::numeric, 0), coalesce((x->>'card')::numeric, 0), coalesce((x->>'credit')::numeric, 0), coalesce((x->>'refunds')::numeric, 0), case when jsonb_typeof(x->'apps') = 'object' then x->'apps' else '{}'::jsonb end, now() from jsonb_array_elements(coalesce(p_rows, '[]')) x where coalesce(x->>'branch', '') <> '' and coalesce(x->>'shift_id', '') <> '' and x->>'bdate' is not null on conflict (branch, shift_id) do update set bdate = excluded.bdate, user_id = excluded.user_id, cashier = excluded.cashier, opened = excluded.opened, closed = excluded.closed, bills = excluded.bills, net = excluded.net, cash = excluded.cash, card = excluded.card, credit = excluded.credit, refunds = excluded.refunds, apps = excluded.apps, synced_at = now(); return jsonb_build_object('ok', true, 'rows', n); end $$; create or replace function public.growth_shifts_for(p_branch text, p_date date) returns jsonb language sql stable security definer set search_path = public as $$ select case when not coalesce(handover_can(p_branch), false) then '[]'::jsonb else coalesce(( select jsonb_agg(jsonb_build_object('shift_id', s.shift_id, 'bdate', s.bdate, 'user_id', s.user_id, 'cashier', s.cashier, 'opened', s.opened, 'closed', s.closed, 'bills', s.bills, 'net', s.net, 'cash', s.cash, 'card', s.card, 'credit', s.credit, 'refunds', s.refunds, 'apps', s.apps, 'code', m.code) order by s.opened nulls last) from growth_shifts s left join growth_cashier_map m on m.branch = s.branch and m.user_id = s.user_id where s.branch = p_branch and (s.bdate = p_date or (s.opened at time zone 'Asia/Riyadh')::date = p_date)), '[]'::jsonb) end $$;
