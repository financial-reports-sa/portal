create or replace function public.clean_alerts_run() returns int language plpgsql volatile security definer set search_path = public as $$ declare d date := att_bday(); b record; sh text; due timestamptz; n int := 0; u uuid[]; lbl text; bn text; nl text := chr(10); late int; begin   if to_regprocedure('public.alert_send(text,text,uuid,uuid[],text,text,text,text,boolean,uuid[])') is null and to_regprocedure('public.alert_send(text,text,uuid,uuid[],text,text,text,text,boolean)') is null then return 0; end if;   for b in select br.id, br.name from branches br            where exists (select 1 from user_sections us join user_branches ub on ub.user_id = us.user_id and ub.branch_id = br.id                          join profiles p on p.id = us.user_id and coalesce(p.active, true) and p.role = 'user' where us.section = 'clean') loop     u := alert_mgrs(b.id, array['clean']);     if cardinality(u) = 0 then continue; end if;     foreach sh in array array['open', 'mid', 'close'] loop       if sh = 'weekly' then         if extract(dow from d) <> 4 then continue; end if;         if exists (select 1 from clean_checks where branch = b.id and shift = 'weekly' and clean_week(day) = clean_week(d)) then continue; end if;       else         if exists (select 1 from clean_checks where branch = b.id and shift = sh and day = d) then continue; end if;       end if;       select ((d + c.due) + case when c.due < time '05:00' then interval '1 day' else interval '0' end) at time zone 'Asia/Riyadh' into due from clean_cfg c where c.shift = sh;       if due is null or now() < due then continue; end if;       if due < coalesce((select since from alert_cfg where id = 1), now()) then continue; end if;       if d <= ((select min(created_at) from clean_items) at time zone 'Asia/Riyadh' - interval '5 hours')::date then continue; end if;       late := floor(extract(epoch from (now() - due)) / 60)::int;       lbl := case sh when 'open' then 'قائمة نظافة الفتح' when 'mid' then 'قائمة نظافة منتصف اليوم' when 'close' then 'قائمة نظافة الإغلاق' else 'التنظيف الأسبوعي' end;       if alert_send('clean:' || b.id || ':' || sh || ':' || d, 'clean_late', null, u, 'إشعار رسمي: ' || lbl || ' لم تُسجَّل',         'السيد/ مدير فرع ' || b.name || ' المحترم،' || nl || nl ||         'انتهى موعد ' || lbl || ' الساعة ' || alert_t12(due) || ' ولم يتم تسجيلها حتى الآن.' || nl || nl ||         'النظافة مسؤوليتكم المباشرة، وهي من أهم معايير تقييم أدائكم وإدارتكم للفرع. والتأخر في تسجيلها يُعد تقصيراً يُسجَّل عليكم ويُرفع للإدارة.' || nl || nl ||         'نأمل المبادرة بإنجاز القائمة وتسجيلها فوراً مع الصور المطلوبة.' || nl || '— إدارة العمليات',         './?tab=clean', 'high', true) then n := n + 1; end if;       if late >= 120 then         perform alert_send('clean2:' || b.id || ':' || sh || ':' || d, 'clean_late2', null, u || alert_owners(), 'إشعار تقصير رسمي: ' || lbl || ' متأخرة أكثر من ساعتين',           'السيد/ مدير فرع ' || b.name || '،' || nl || nl ||           'مضى أكثر من ساعتين على موعد ' || lbl || ' (' || alert_t12(due) || ') ولم تُسجَّل حتى الآن رغم التنبيه السابق.' || nl || nl ||           'تم تسجيل ذلك تقصيراً في سجل أداء الفرع ورفعه للإدارة. نذكّر بأن الالتزام بالنظافة معيار أساسي في تقييم إدارة الفرع، واستمرار التقصير يستوجب اتخاذ الإجراء الإداري اللازم.' || nl || nl ||           'المطلوب تسجيل القائمة فوراً.' || nl || '— نظام المتابعة الآلي', './?tab=clean', 'urgent', true);       end if;     end loop;   end loop;   return n; end $$; revoke all on function public.clean_alerts_run() from public, anon, authenticated;  create or replace function public.clean_ai_save(p_id uuid, p_items jsonb, p_note text, p_err text default '') returns int language plpgsql volatile security definer set search_path = public as $$ declare c clean_checks;     return null;   end if;   select round(avg(case when coalesce((it->>'ok')::boolean, false)                         then greatest(1, least(5, coalesce((select (a->>'score')::numeric from jsonb_array_elements(p_items) a where (a->>'i')::int = (q.n - 1)::int limit 1), 3)))                         else 1 end))::int     into r   from jsonb_array_elements(c.items) with ordinality q(it, n);   end if;   if r <= 3 then     select coalesce(array_agg(id), '{}') into o from profiles where role = 'owner' and coalesce(active, true);     perform push_send(o, '🤖 نظافة ' || bn || ': ' || r || ' من 5', left(coalesce(nullif(p_note, ''), 'تقييم منخفض للنظافة'), 220), './?tab=clean', 'clean-ai-low-' || p_id);   end if;   return r; end $$; revoke all on function public.clean_ai_save(uuid, jsonb, text, text) from public, anon, authenticated; grant execute on function public.clean_ai_save(uuid, jsonb, text, text) to service_role;
