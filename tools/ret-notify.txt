create or replace function public.ret_photo(p_id bigint) returns text language sql stable security definer set search_path = public as $$ select p.img from cashier_return_photos p join cashier_returns r on r.id = p.ret_id where p.ret_id = p_id and ret_view_can(r.branch) $$; end if; mins := floor(extract(epoch from (nr - cand)) / 60); if mins > 30 then return jsonb_build_object('ok', false, 'error', 'late', 'message', 'انتهت مهلة التوثيق — المرتجع لازم يتوثّق خلال نص ساعة من وقت الفاتورة (مرّ ' || mins || ' دقيقة). المرتجع غير الموثّق يُحسب عجز عليك.'); end if; v_at := cand at time zone 'Asia/Riyadh'; v_bd := (nr - interval '6 hours')::date; if exists (select 1 from cashier_returns where branch = p_branch and bdate = v_bd and inv_no = v_inv) then return jsonb_build_object('ok', false, 'message', 'مرتجع هذه الفاتورة موثّق قبل');
