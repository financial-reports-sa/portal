old_role text; ph text; other uuid; begin   if not coalesce(is_admin(), false) then raise exception 'الدمج للإدارة فقط'; end if;   select * into e from employees where id = p_emp;   if e.id is null then raise exception 'الموظف غير موجود'; end if;   select * into pr from profiles where id = p_user;   if pr.id is null or pr.role not in ('user', 'admin', 'owner') then raise exception 'اختر حساب مستخدم صحيح'; end if;   select emp_id into other from emp_accounts where user_id = p_user and emp_id <> p_emp;       update msg_replies set user_id = p_user where user_id = old_uid;       update msg_replies set to_user = p_user where to_user = old_uid;       delete from auth.users where id = old_uid;     else       delete from emp_accounts where emp_id = p_emp;     end if;   end if;   if coalesce(ph, '') = '' or exists (select 1 from emp_accounts where phone = ph) then ph := 'u' || left(replace(p_user::text, '-', ''), 12); end if;   insert into emp_accounts (emp_id, user_id, phone, created_by) values (p_emp, p_user, ph, who_am_i());
